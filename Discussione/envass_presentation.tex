	\begin{frame}{Contesto: Evoluzione di ENVASS}
		
		\textbf{CIAS~\cite{cias} – Centro di Ricerca dell'Università di Ferrara}
		
		\vspace{0.2cm}
		
		ENVASS 2.0 nasce come evoluzione del sistema gestionale esistente
		utilizzato per la gestione e il tracciamento di campioni ambientali
		in ambienti ad alta sterilità.
		
		
		\vspace{0.4cm}
		
		\begin{columns}[T]
			
			\begin{column}{0.48\textwidth}
				\begin{alertblock}{ENVASS 1.0 (Origini)}
					\begin{itemize}
						\item Interfaccia web statica
						\item Catalogazione e inserimento foto manuali
						\item Estrazione dati limitata a filtri base (progetto, struttura, date)
						\item Mancanza di controllo sull'intera filiera operativa
					\end{itemize}
				\end{alertblock}
			\end{column}
			
			\begin{column}{0.48\textwidth}
				\begin{block}{ENVASS 2.0}
					\begin{itemize}
						\item Workflow completamente digitalizzato
						\item Generazione automatica dei campioni
						\item Audit centralizzato
						\item Integrità e immutabilità dei dati
					\end{itemize}
				\end{block}
			\end{column}
			
		\end{columns}
		
	\end{frame}
